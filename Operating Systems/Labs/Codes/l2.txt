#include<iostream>
#include<stdlib.h>

using namespace std;

int main(int argc, char *argv[])
{
    int sum = 0;

    for(int i = 1; i < argc; i++)
    {
        sum += atoi(argv[i]);
    }

    int average = sum / (argc - 1);

    cout << "Sum= " << sum << endl;
    cout << "Average= " << average << endl;

    return 0;
}

#include<iostream>
#include<stdlib.h>

using namespace std;

int main(int argc, char *argv[])
{
    int size = argc - 1;
    int array[size];

    for(int i = 0; i < size; i++)
    {
        array[i] = atoi(argv[i + 1]);
    }

    for(int i = 0; i < size; i++)
    {
        for(int j = i + 1; j < size; j++)
        {
            if(array[i] > array[j])
            {
                int temp = array[i];
                array[i] = array[j];
                array[j] = temp;
            }
        }
    }

    cout << "Sorted Array= ";

    for(int i = 0; i < size; i++)
    {
        cout << array[i] << " ";
    }

    return 0;
}

#include<stdio.h>

int sum_array(int arr[], int size)
{
    int sum = 0;

    for(int i = 0; i < size; i++)
    {
        sum += arr[i];
    }

    return sum;
}
