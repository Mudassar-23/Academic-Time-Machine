#include <stdio.h> 
#include <string.h> 
#include <arpa/inet.h> 
#include <sys/socket.h> 
#include <unistd.h> 
 
int main() 
{ 
    int sd, new_sd; 
    char city[100]; 
    char response[100]; 
 
    struct sockaddr_in my_addr, client_addr; 
    socklen_t size; 
 
    my_addr.sin_family = AF_INET; 
    my_addr.sin_port = htons(4999); 
    my_addr.sin_addr.s_addr = inet_addr("127.0.0.1"); 
 
    sd = socket(AF_INET, SOCK_STREAM, 0); 
 
    bind(sd, (struct sockaddr *)&my_addr, sizeof(my_addr)); 
 
    listen(sd, 5); 
 
    size = sizeof(client_addr); 
 
    new_sd = accept(sd, (struct sockaddr *)&client_addr, &size); 
 
    recv(new_sd, city, sizeof(city), 0); 
 
    printf("Client asked weather for: %s\n", city); 
 
    if(strcmp(city, "Karachi") == 0) 
        strcpy(response, "Karachi: 25C, Sunny"); 
    else if(strcmp(city, "Lahore") == 0) 
        strcpy(response, "Lahore: 30C, Hot"); 
    else if(strcmp(city, "Islamabad") == 0) 
        strcpy(response, "Islamabad: 22C, Cloudy"); 
    else 
        strcpy(response, "City not found"); 
 
    send(new_sd, response, strlen(response)+1, 0); 
 
    close(new_sd); 
close(sd); 
return 0; 
}
